local function getRequestFunction()

        return syn.request
    end

local function sendWebhook(eventName, details)
    if not WEBHOOK_LOGGING or not GAME_LOADERS[game.PlaceId] then
        return
    end

    local gameName = GAME_LOADERS[game.PlaceId].Name
    local webhook = WEBHOOKS[gameName]
    if type(webhook) ~= "string" or webhook == "" then
        return
    end

    local requestFn = getRequestFunction()
    if not requestFn then
        return
    end

    local fields = {
        {name = "Player", value = tostring(LocalPlayer and (LocalPlayer.Name .. " (" .. LocalPlayer.UserId .. ")") or "Unknown"), inline = false},
        {name = "Game", value = gameName, inline = true},
        {name = "PlaceId", value = tostring(game.PlaceId), inline = true},
        {name = "Event", value = tostring(eventName), inline = true},
        {name = "Executor", value = getExecutorName(), inline = true},
    }

    for key, value in pairs(details or {}) do
        table.insert(fields, {
            name = tostring(key),
            value = tostring(value):sub(1, 1000),
            inline = false,
        })
    end

    local payload = {
        username = "ZeHub Logger",
        embeds = {{
            title = "ZeHub • " .. tostring(eventName),
            description = "FlowAuth loader event",
            color = 16737792,
            fields = fields,
            footer = {text = "ZeHub • FlowAuth"},
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
        }},
    }

    pcall(function()
        requestFn({
            Url = webhook,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
            },
            Body = HttpService:JSONEncode(payload),
        })
    end)
end

local function clipboard(text)
    pcall(function()
        if type(setclipboard) == "function" then
            setclipboard(text)
        elseif type(toclipboard) == "function" then
            toclipboard(text)
        end
    end)
end

getExecutorName = function()
    local ok, result = pcall(function()
        if type(identifyexecutor) == "function" then
            return identifyexecutor()
        end
        return "Unknown"
    end)

    if not ok then
        return "Unknown"
    end

    if type(result) == "table" then
        return tostring(result.Name or result[1] or "Unknown")
    end

    return tostring(result or "Unknown")
end

local function isLowCompatibilityExecutor()
    local name = getExecutorName():lower()
    return name:find("xeno", 1, true)
        or name:find("solara", 1, true)
        or name:find("jjsploit", 1, true)
end

local function getHttp(url)
    local ok, result = pcall(function()
        return game:HttpGet(url)
    end)

    if not ok or type(result) ~= "string" or result == "" then
        return nil, tostring(result or "Empty response")
    end

    return result
end

local function loadFlowAuthProject(hash, key)
    local url = "https://flowauth.net/v1/loaders/" .. hash .. ".lua"
    local source, err = getHttp(url)

    if not source then
        return false, "Unable to reach FlowAuth. " .. tostring(err)
    end

    if type(loadstring) ~= "function" then
        return false, "Your executor does not support loadstring."
    end

    local compiled, compileError = loadstring(source)
    if type(compiled) ~= "function" then
        return false, "FlowAuth loader failed to compile: " .. tostring(compileError)
    end

    -- IMPORTANT: FlowAuth's generated loader expects the customer/project key
    -- as its first argument. The previous version called compiled() with no
    -- argument, which caused the "premium_required / valid project key" error.
    local ok, result = pcall(compiled, key)
    if not ok then
        return false, "FlowAuth loader failed: " .. tostring(result)
    end

    -- Support projects that return a second-stage validator.
    if type(result) == "function" then
        local verifiedOK, verified = pcall(result, key)
        if not verifiedOK then
            return false, "FlowAuth validation failed: " .. tostring(verified)
        end
        return verified ~= false, verified
    end

    return result ~= false, result
end

--// Key storage --------------------------------------------------------------
local KEY_FILE = "ZeHub_Key.txt"

local function readSavedKey()
    if type(isfile) ~= "function" or type(readfile) ~= "function" then
        return ""
    end

    local ok, exists = pcall(isfile, KEY_FILE)
    if not ok or not exists then
        return ""
    end

    local readOK, value = pcall(readfile, KEY_FILE)
    if not readOK then
        return ""
    end

    return tostring(value or ""):gsub("^%s+", ""):gsub("%s+$", "")
end

local function saveKey(value)
    if type(writefile) ~= "function" then
        return false
    end

    local ok = pcall(writefile, KEY_FILE, tostring(value or ""))
    return ok
end

local function clearSavedKey()
    if type(delfile) == "function" then
        pcall(delfile, KEY_FILE)
    elseif type(writefile) == "function" then
        pcall(writefile, KEY_FILE, "")
    end
end

-- Load is intentionally first, followed by Get Key and Discord.
KeySection:CreateButton({
    Name = "Load Script",
    Callback = function()
        if busy then
            return
        end

        if not GAME then
            setStatus("Unsupported game", Color3.fromRGB(240, 170, 90))
            return
        end

        if isLowCompatibilityExecutor() then
            setStatus("Executor may not be compatible", Color3.fromRGB(240, 170, 90))
            warn("[ZeHub] Executor may not be compatible: " .. getExecutorName())
            return
        end

        local key = tostring(currentKey or ""):gsub("^%s+", ""):gsub("%s+$", "")

        if key == "" then
            setStatus("Enter your key first", Color3.fromRGB(240, 120, 120))
            return
        end

        busy = true
        setStatus("Authenticating key...", Color3.fromRGB(235, 190, 80))
        sendWebhook("Button Clicked", {Action = "Load Script"})
        sendWebhook("Validation Started")

        local valid, result = loadFlowAuthProject(GAME.Hash, key)

        if not valid then
            setStatus("Authentication failed", Color3.fromRGB(240, 100, 100))
            warn("[ZeHub] FlowAuth rejected the key: " .. tostring(result))
            sendWebhook("Validation Failed", {
                Reason = tostring(result):sub(1, 500)
            })
            busy = false
            return
        end

        -- Only remember the key after FlowAuth accepts it.
        saveKey(key)
        currentKey = key

        setStatus("Key accepted • Loading script...", Color3.fromRGB(100, 220, 140))
        sendWebhook("Validation Success")

        task.wait(0.15)

        pcall(function()
            Window:Destroy()
        end)

        busy = false
    end,
})
