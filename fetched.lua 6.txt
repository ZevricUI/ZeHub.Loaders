-- Find all balls function
local function FindAllBalls()
    trackedBalls = {}
    
    local tpsSystem = workspace:FindFirstChild("TPSSystem")
    if tpsSystem then
        local ball = tpsSystem:FindFirstChild("TPS")
        if ball and ball:IsA("BasePart") then
            table.insert(trackedBalls, ball)
        end
    end
    
    local practice = workspace:FindFirstChild("Practice")
    if practice then
        for _, child in ipairs(practice:GetChildren()) do
            if child:IsA("BasePart") and child.Name == "PSoccerBall" then
                table.insert(trackedBalls, child)
            end
        end
    end
    
    -- Also check workspace directly for any other balls
    for _, child in ipairs(workspace:GetChildren()) do
        if child:IsA("BasePart") and (child.Name == "TPS" or child.Name == "PSoccerBall") then
            local alreadyTracked = false
            for _, ball in ipairs(trackedBalls) do
                if ball == child then
                    alreadyTracked = true
                    break
                end
            end
            if not alreadyTracked then
                table.insert(trackedBalls, child)
            end
        end
    end
end

-- Get correct leg based on UI toggle
local function GetCorrectLeg()
    local char = LocalPlayer.Character
    if not char then return nil end
    
    local leg = char:FindFirstChild("Right Leg")
    local gui = LocalPlayer.PlayerGui
    if gui then
        local startGui = gui:FindFirstChild("Start")
        if startGui and startGui:FindFirstChild("Points") then
            local legSwitch = startGui.Points:FindFirstChild("Leg")
            if legSwitch and legSwitch:FindFirstChild("Switch") then
                if legSwitch.Switch.Text == "L" then
                    leg = char:FindFirstChild("Left Leg")
                end
            end
        end
    end
    return leg
end

-- FireTouch function for all body parts
local function FireTouchAll()
    local char = LocalPlayer.Character
    if not char then return end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    -- Update balls constantly
    FindAllBalls()
    
    if #trackedBalls == 0 then return end
    
    -- Find nearest ball
    local nearestBall = nil
    local nearestDist = ReachDistance
    
    for _, ball in ipairs(trackedBalls) do
        if ball and ball.Parent then
            local dist = (root.Position - ball.Position).Magnitude
            if dist <= nearestDist then
                nearestDist = dist
                nearestBall = ball
            end
        end
    end
    
    if not nearestBall then return end
    
    -- All body parts
    local bodyParts = {
        "Right Leg", "Left Leg", "Right Arm", 
        "Left Arm", "Head", "Torso", "HumanoidRootPart"
    }
    
    for _, partName in ipairs(bodyParts) do
        local part = char:FindFirstChild(partName)
        if part then
            pcall(function()
                firetouchinterest(part, nearestBall, 0)
                firetouchinterest(part, nearestBall, 1)
            end)
        end
    end
end

-- Enable FireTouch
local function EnableFireTouch()
    if FireTouchEnabled then return end
    FireTouchEnabled = true
    
    -- Start connections for all limbs (yBloodz style - each limb gets its own loop)
    local limbNames = {"Right Leg", "Left Leg", "Right Arm", "Left Arm", "Head", "Torso", "HumanoidRootPart"}
    
    for _, limbName in ipairs(limbNames) do
        local conn = RunService.RenderStepped:Connect(function()
            if not FireTouchEnabled then return end
            
            local char = LocalPlayer.Character
            if not char then return end
            
            local limb = char:FindFirstChild(limbName)
            if not limb then return end
            
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            
            -- Update balls
            FindAllBalls()
            
            if #trackedBalls == 0 then return end
            
            -- Find nearest ball
            local nearestBall = nil
            local nearestDist = ReachDistance
            
            for _, ball in ipairs(trackedBalls) do
                if ball and ball.Parent then
                    local dist = (root.Position - ball.Position).Magnitude
                    if dist <= nearestDist then
                        nearestDist = dist
                        nearestBall = ball
                    end
                end
            end
            
            if nearestBall then
                pcall(function()
                    firetouchinterest(limb, nearestBall, 0)
                    firetouchinterest(limb, nearestBall, 1)
                end)
            end
        end)
        table.insert(connections, conn)
    end
    
    StatusLabel.Text = "🟢 ON"
    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
    ToggleButton.Text = "■ STOP"
    ToggleButton.BackgroundColor3 = Color3.fromRGB(200, 60, 60)
end

-- Disable FireTouch
local function DisableFireTouch()
    FireTouchEnabled = false
    
    for _, conn in ipairs(connections) do
        pcall(function()
            conn:Disconnect()
        end)
    end
    connections = {}
    
    StatusLabel.Text = "⚪ OFF"
    StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
    ToggleButton.Text = "▶ START"
    ToggleButton.BackgroundColor3 = Color3.fromRGB(60, 60, 70)
end

-- Toggle function
local function ToggleFireTouch()
    if FireTouchEnabled then
        DisableFireTouch()
    else
        EnableFireTouch()
    end
end

-- Button callbacks
ToggleButton.MouseButton1Click:Connect(ToggleFireTouch)

-- Minimize functionality

    MainFrame.Visible = not minimized
    MinimizedFrame.Visible = minimized
    MinimizeButton.Text = minimized and "+" or "−"

ExpandButton.MouseButton1Click:Connect(function()
    minimized = false
    MainFrame.Visible = true
    MinimizedFrame.Visible = false
    MinimizeButton.Text = "−"
end)

-- Distance input
DistanceBox.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        local dist = tonumber(DistanceBox.Text)
        if dist and dist > 0 then
            ReachDistance = dist
        else
            DistanceBox.Text = tostring(ReachDistance)
        end
    end
end)

-- Auto-find balls when they appear
workspace.DescendantAdded:Connect(function(desc)
    if FireTouchEnabled and desc:IsA("BasePart") and (desc.Name == "TPS" or desc.Name == "PSoccerBall") then
        local alreadyTracked = false
        for _, ball in ipairs(trackedBalls) do
            if ball == desc then
                alreadyTracked = true
                break
            end
        end
        if not alreadyTracked then
            table.insert(trackedBalls, desc)
        end
    end
end)

workspace.DescendantRemoved:Connect(function(desc)
    if FireTouchEnabled and desc:IsA("BasePart") and (desc.Name == "TPS" or desc.Name == "PSoccerBall") then
        for i, ball in ipairs(trackedBalls) do
            if ball == desc then
                table.remove(trackedBalls, i)
                break
            end
        end
    end
end)
